Ejemplo de código no compatible (AsyncAPI 2)

asyncapi: '2.4.0'
info:
  title: Mi API Asíncrona
  version: '1.0.0'
channels:
  pet/created:
    subscribe:   # Noncompliant {{AAR051: El operationId debe estar presente y seguir la convención camelCase}}
      summary: Recibe eventos de mascota creada.
      message:
        payload:
          type: object

Solución compatible (AsyncAPI 2)

asyncapi: '2.4.0'
info:
  title: Mi API Asíncrona
  version: '1.0.0'
channels:
  pet/created:
    subscribe:
      operationId: solicitarBeca
      summary: Recibe eventos de mascota creada.
      message:
        payload:
          type: object

Ejemplo de código no compatible (AsyncAPI 3)

asyncapi: '3.0.0'
info:
  title: Mi API Asíncrona
  version: '1.0.0'
channels:
  petCreated:
    address: pet/created
    messages:
      PetCreated:
        payload:
          type: object
operations:
  onPetCreated: # Noncompliant {{AAR051: El operationId debe estar presente y seguir la convención camelCase}}
    action: receive
    channel:
      $ref: '#/channels/petCreated'
    messages:
      - $ref: '#/channels/petCreated/messages/PetCreated'

Solución compatible (AsyncAPI 3)

asyncapi: '3.0.0'
info:
  title: Mi API Asíncrona
  version: '1.0.0'
channels:
  petCreated:
    address: pet/created
    messages:
      PetCreated:
        payload:
          type: object
operations:
  onPetCreated:
    action: receive
    operationId: solicitarBeca
    channel:
      $ref: '#/channels/petCreated'
    messages:
      - $ref: '#/channels/petCreated/messages/PetCreated'
