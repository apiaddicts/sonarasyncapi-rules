Compliant Solution (AsyncAPI 2)

asyncapi: '2.4.0'
info:
  title: Unique Operation ID Example API
  version: '1.0.0'
  description: API demonstrating unique operationId usage in each operation.
channels:
  /pets:
    subscribe:
      operationId: getPets
      summary: Receive the list of pets.
      message:
        contentType: application/json
        payload:
          type: object
