Compliant Solution (AsyncAPI 2)

asyncapi: '2.4.0'
info:
  title: Numeric Parameter Restrictions Example API
  version: '1.0.0'
  description: API de ejemplo que incluye restricciones para parámetros numéricos.

channels:
  /sensor-data:
    description: Canal para publicar datos de sensores.
    publish:
      operationId: sendSensorData
      summary: Publica datos recogidos por un sensor.
      message:
        contentType: application/json
        payload:
          type: object
          properties: 
            humidity:
              type: number
              description: Humedad medida por el sensor.
              minimum: 0
              maximum: 100
            pressure:
              type: number
              description: Presión atmosférica medida.
              minimum: 900
              maximum: 1100

Compliant Solution (AsyncAPI 3)

asyncapi: '3.0.0'
info:
  title: Numeric Parameter Restrictions Example API
  version: '1.0.0'
  description: Example API that includes restrictions for numeric parameters.
servers:
  production:
    host: broker.example.com
    protocol: kafka
channels:
  sensorData:
    address: sensor-data
    messages:
      SensorData:
        contentType: application/json
        payload:
          type: object
          properties:
            humidity:
              type: number
              description: Humidity measured by the sensor.
              minimum: 0
              maximum: 100
            pressure:
              type: number
              description: Atmospheric pressure measured.
              minimum: 900
              maximum: 1100
operations:
  sendSensorData:
    action: send
    channel:
      $ref: '#/channels/sensorData'
    messages:
      - $ref: '#/channels/sensorData/messages/SensorData'
