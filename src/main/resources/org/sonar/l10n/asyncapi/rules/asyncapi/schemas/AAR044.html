Avro records must declare a namespace to avoid name collisions and to enable schema evolution across different services and teams.

Noncompliant Code Example (AsyncAPI 2)

message:
  schemaFormat: 'application/vnd.apache.avro;version=1.9.0'
  payload:
    type: record
    name: OrderValue
    fields:
      - name: id
        type: string

Compliant Solution (AsyncAPI 2)

message:
  schemaFormat: 'application/vnd.apache.avro;version=1.9.0'
  payload:
    type: record
    name: OrderValue
    namespace: org.example.orders
    fields:
      - name: id
        type: string

Noncompliant Code Example (AsyncAPI 3)

asyncapi: '3.0.0'
info:
  title: Avro Namespace Required
  version: 1.0.0
channels:
  orders:
    address: orders
    messages:
      OrderMessage:
        payload:
          schemaFormat: 'application/vnd.apache.avro;version=1.9.0'
          schema: # Noncompliant {{AAR044: Avro record must have the namespace field defined}}
            type: record
            name: OrderValue
            fields:
              - name: id
                type: string

Compliant Solution (AsyncAPI 3)

asyncapi: '3.0.0'
info:
  title: Avro Namespace Required
  version: 1.0.0
channels:
  orders:
    address: orders
    messages:
      OrderMessage:
        payload:
          schemaFormat: 'application/vnd.apache.avro;version=1.9.0'
          schema:
            type: record
            name: OrderValue
            namespace: org.example.orders
            fields:
              - name: id
                type: string
