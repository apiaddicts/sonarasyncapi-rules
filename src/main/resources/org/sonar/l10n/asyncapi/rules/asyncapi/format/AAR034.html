Numeric types require a valid format

Noncompliant Code Example (AsyncAPI 2)

asyncapi: '2.4.0'
info:
  version: '1.0.0'
  title: AsyncAPI Petstore
channels:
  /invoices:
    description: Channel for invoices
    publish:
      summary: Publish invoice data
      operationId: publishInvoice
      message:
        contentType: application/json
        payload:
          type: object
          properties:
            price:
              type: number
              format: double
            nested:
              type: object
              properties:
                value:
                  type: number # Noncompliant {{AAR034: Numeric property must specify a valid 'format' (int32, int64, float, double).}}

Compliant Solution (AsyncAPI 2)

asyncapi: '2.4.0'
info:
  version: '1.0.0'
  title: AsyncAPI Petstore
channels:
  /invoices:
    description: Channel for invoices
    publish:
      summary: Publish invoice data
      operationId: publishInvoice
      message:
        contentType: application/json
        payload:
          type: object
          properties:
            price:
              type: number
              format: double
            nested:
              type: object
              properties:
                value:
                  type: number
                  format: int64

info:
  version: '1.0.0'
  title: AsyncAPI Petstore
servers:
  production:
    host: broker.example.com
    protocol: kafka
channels:
  invoices:
    address: invoices
    messages:
      Invoice:
        contentType: application/json
        payload:
          type: object
          properties:
            price:
              type: number
              format: double
            nested:
              type: object
              properties:
                value:
                  type: number
                  format: int64
operations:
  publishInvoice:
    action: send
    channel:
      $ref: '#/channels/invoices'
    messages:
      - $ref: '#/channels/invoices/messages/Invoice'
