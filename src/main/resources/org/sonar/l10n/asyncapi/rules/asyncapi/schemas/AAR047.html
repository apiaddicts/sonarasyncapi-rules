Avro fields should include a doc attribute to describe the purpose of each field. This improves schema discoverability and consumer understanding.

Noncompliant Code Example (AsyncAPI 2)

payload:
  type: record
  name: OrderValue
  namespace: org.example.orders
  fields:
    - name: id
      type: string

Compliant Solution (AsyncAPI 2)

payload:
  type: record
  name: OrderValue
  namespace: org.example.orders
  fields:
    - name: id
      type: string
      doc: "The unique order identifier"

Noncompliant Code Example (AsyncAPI 3)

asyncapi: '3.0.0'
info:
  title: Avro Field Doc
  version: 1.0.0
channels:
  orders:
    address: orders
    messages:
      OrderMessage:
        payload:
          schemaFormat: 'application/vnd.apache.avro;version=1.9.0'
          schema: # Noncompliant {{AAR047: Avro field should include the doc attribute with a description}}
            type: record
            name: OrderValue
            namespace: org.example.orders
            fields:
              - name: id
                type: string

Compliant Solution (AsyncAPI 3)

asyncapi: '3.0.0'
info:
  title: Avro Field Doc
  version: 1.0.0
channels:
  orders:
    address: orders
    messages:
      OrderMessage:
        payload:
          schemaFormat: 'application/vnd.apache.avro;version=1.9.0'
          schema:
            type: record
            name: OrderValue
            namespace: org.example.orders
            fields:
              - name: id
                type: string
                doc: "The unique order identifier"
