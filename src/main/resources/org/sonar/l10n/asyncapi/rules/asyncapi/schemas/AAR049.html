When an Avro field uses a nullable union type (e.g. ["null", "string"]), it must define default: null. Without a null default, consumers that were deployed before the field was added will fail to deserialize messages, breaking backward compatibility.

Noncompliant Code Example (AsyncAPI 2)

payload:
  type: record
  name: OrderValue
  namespace: org.example.orders
  fields:
    - name: description
      type:
        - "null"
        - string

Compliant Solution (AsyncAPI 2)

payload:
  type: record
  name: OrderValue
  namespace: org.example.orders
  fields:
    - name: description
      type:
        - "null"
        - string
      default: null

Noncompliant Code Example (AsyncAPI 3)

asyncapi: '3.0.0'
info:
  title: Avro Optional Field Default
  version: 1.0.0
channels:
  orders:
    address: orders
    messages:
      OrderMessage:
        payload:
          schemaFormat: 'application/vnd.apache.avro;version=1.9.0'
          schema: # Noncompliant {{AAR049: Optional Avro fields (union with null) must declare their default value as null}}
            type: record
            name: OrderValue
            namespace: org.example.orders
            fields:
              - name: id
                type: string
              - name: description
                type:
                  - "null"
                  - string

Compliant Solution (AsyncAPI 3)

asyncapi: '3.0.0'
info:
  title: Avro Optional Field Default
  version: 1.0.0
channels:
  orders:
    address: orders
    messages:
      OrderMessage:
        payload:
          schemaFormat: 'application/vnd.apache.avro;version=1.9.0'
          schema:
            type: record
            name: OrderValue
            namespace: org.example.orders
            fields:
              - name: id
                type: string
              - name: description
                type:
                  - "null"
                  - string
                default: null
