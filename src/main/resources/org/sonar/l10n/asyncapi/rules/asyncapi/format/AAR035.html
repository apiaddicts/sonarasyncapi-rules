It is recommended to have a title per message

Noncompliant Code Example (AsyncAPI 2)

asyncapi: '2.4.0'
info:
  title: Message Title Example API
  version: '1.0.0'
  description: API de ejemplo que incluye un título por cada mensaje.

channels:
  /order:
    description: Canal para enviar y recibir información de pedidos.
    publish:
      operationId: sendOrder
      summary: Publica información sobre un nuevo pedido.
      message: #Noncompliant {{AAR035: Message should have a 'title' field.}}
        description: Mensaje que contiene los detalles de un nuevo pedido realizado.
        contentType: application/json
        payload:
          type: object
          properties:
            orderId:
              type: string
              description: ID único del pedido.
            product:
              type: string
              description: Nombre del producto.
            quantity:
              type: integer
              description: Cantidad del producto pedido.
    subscribe:
      operationId: receiveOrderConfirmation
      summary: Suscripción para recibir confirmación de pedidos.
      message:
        title: Confirmación de Pedido
        description: Mensaje que contiene la confirmación de un pedido.
        contentType: application/json
        payload:
          type: object
          properties:
            orderId:
              type: string
              description: ID único del pedido.
            status:
              type: string
              description: Estado de la confirmación del pedido.

Compliant Solution (AsyncAPI 2)

asyncapi: '2.4.0'
info:
  title: Message Title Example API
  version: '1.0.0'
  description: API de ejemplo que incluye un título por cada mensaje.

channels:
  /order:
    description: Canal para enviar y recibir información de pedidos.
    publish:
      operationId: sendOrder
      summary: Publica información sobre un nuevo pedido.
      message: 
        title: message title
        description: Mensaje que contiene los detalles de un nuevo pedido realizado.
        contentType: application/json
        payload:
          type: object
          properties:
            orderId:
              type: string
              description: ID único del pedido.
            product:
              type: string
              description: Nombre del producto.
            quantity:
              type: integer
              description: Cantidad del producto pedido.
    subscribe:
      operationId: receiveOrderConfirmation
      summary: Suscripción para recibir confirmación de pedidos.
      message:
        title: Confirmación de Pedido
        description: Mensaje que contiene la confirmación de un pedido.
        contentType: application/json
        payload:
          type: object
          properties:
            orderId:
              type: string
              description: ID único del pedido.
            status:
              type: string
              description: Estado de la confirmación del pedido.

Compliant Solution (AsyncAPI 3)

asyncapi: '3.0.0'
info:
  title: Message Title Avro Example API
  version: 1.0.0
servers:
  production:
    host: api.example.com
    protocol: kafka
channels:
  carga:
    address: carga
    messages:
      CargaMessage:
        title: Carga Event
        payload:
          schemaFormat: 'application/vnd.apache.avro;version=1.9.0'
          schema:
            type: record
            name: CargaValue
            namespace: com.example.avro
            fields:
              - name: id
                type: string
