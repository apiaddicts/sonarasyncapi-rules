Compliant Solution (AsyncAPI 3)

asyncapi: '3.0.0'
info:
  title: Unique Operation ID Example API
  version: '1.0.0'
  description: API that demonstrates the use of a unique operationId in each operation.
servers:
  production:
    host: api.example.com
    protocol: kafka
channels:
  pets:
    address: pets
    description: Hello.
    messages:
      Pet:
        contentType: application/json
        payload:
          type: object
          properties:
            id:
              type: string
operations:
  getPets:
    action: receive
    description: Hello.
    channel:
      $ref: '#/channels/pets'
    messages:
      - $ref: '#/channels/pets/messages/Pet'
