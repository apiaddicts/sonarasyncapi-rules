Avro names (for records and fields) must start with [A-Za-z_] and subsequently contain only [A-Za-z0-9_]. Names starting with digits or containing hyphens are invalid and will cause schema parsing failures.

Noncompliant Code Example (AsyncAPI 2)

payload:
  type: record
  name: 0-invalid-record
  namespace: org.example.orders
  fields:
    - name: order-id
      type: string

Compliant Solution (AsyncAPI 2)

payload:
  type: record
  name: OrderValue
  namespace: org.example.orders
  fields:
    - name: orderId
      type: string

Noncompliant Code Example (AsyncAPI 3)

asyncapi: '3.0.0'
info:
  title: Avro Name Nomenclature
  version: 1.0.0
channels:
  orders:
    address: orders
    messages:
      OrderMessage:
        payload:
          schemaFormat: 'application/vnd.apache.avro;version=1.9.0'
          schema: # Noncompliant {{AAR048: Avro name must start with a letter or underscore, and can only contain letters, numbers and underscores}}
            type: record
            name: 0-invalid-record
            namespace: org.example.orders
            fields:
              - name: id
                type: string

Compliant Solution (AsyncAPI 3)

asyncapi: '3.0.0'
info:
  title: Avro Name Nomenclature
  version: 1.0.0
channels:
  orders:
    address: orders
    messages:
      OrderMessage:
        payload:
          schemaFormat: 'application/vnd.apache.avro;version=1.9.0'
          schema:
            type: record
            name: OrderValue
            namespace: org.example.orders
            fields:
              - name: orderId
                type: string
