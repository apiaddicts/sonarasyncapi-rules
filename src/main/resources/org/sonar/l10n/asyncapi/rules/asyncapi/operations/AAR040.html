Channel server must be defined in the servers object

Noncompliant Code Example (AsyncAPI 2)

asyncapi: '2.4.0'
info:
  title: Swagger Petstore
  version: '1.0.0'
servers:
  production:
    url: 'https://api.petstore.com'
    protocol: 'https'
    description: Servidor de producción para la API de la tienda de mascotas
channels: # Noncompliant {{AAR040: Channel server must be defined in the servers object}}
    /pets:
      description: Canal de comunicación para operaciones relacionadas con mascotas
      subscribe: 
        operationId: receivePets
        summary: Operación para recibir la lista de mascotas
        message:
          contentType: application/json
          payload:
            type: array
            items:
              type: object
              properties:
                id:
                  type: string
                  description: ID de la mascota
                name:
                  type: string
                  description: Nombre de la mascota

Compliant Solution (AsyncAPI 2)

asyncapi: "2.0.0"
info:
  title: Awesome API
  description: A very well-defined API
  version: "1.0"
servers:
  production:
    url: "stoplight.io"
    protocol: "https"
channels:
  hello:
    servers: production

asyncapi: '3.0.0'
info:
  title: Invalid Server Reference

servers:
  production:
    host: api.example.com
    protocol: wss
channels:

Compliant Solution (AsyncAPI 3)

asyncapi: '3.0.0'
info:
  title: Valid Server Reference
  version: '1.0.0'
servers:
  production:
    host: api.example.com
    protocol: wss
channels:
  notifications:
    address: notifications
    servers:
      - production
    messages:
      Notification:
        payload:
          type: object
operations:
  SendNotification:
    action: send
    channel:
      $ref: '#/channels/notifications'
    messages:
      - $ref: '#/channels/notifications/messages/Notification'
