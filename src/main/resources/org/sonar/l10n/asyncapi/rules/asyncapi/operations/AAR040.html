Channel server must be defined in the servers object

Noncompliant Code Example (AsyncAPI 2)

asyncapi: '2.4.0'
info:
  title: Swagger Petstore
  version: '1.0.0'
servers:
  production:
    url: 'https://api.petstore.com'
    protocol: 'https'
channels:
  /pets:
    description: Channel for pet-related operations
    servers:
      - unknown # Noncompliant {{AAR040: Channel references a server that is not defined in the 'servers' object.}}
    subscribe:
      operationId: receivePets
      message:
        contentType: application/json
        payload:
          type: object

Compliant Solution (AsyncAPI 2)

asyncapi: '2.4.0'
info:
  title: Swagger Petstore
  version: '1.0.0'
servers:
  production:
    url: 'https://api.petstore.com'
    protocol: 'https'
channels:
  /pets:
    description: Channel for pet-related operations
    servers:
      - production
    subscribe:
      operationId: receivePets
      message:
        contentType: application/json
        payload:
          type: object

Noncompliant Code Example (AsyncAPI 3)

asyncapi: '3.0.0'
info:
  title: Invalid Server Reference
  version: '1.0.0'
servers:
  production:
    host: api.example.com
    protocol: wss
channels:
  notifications:
    address: notifications
    servers:
      - unknown # Noncompliant {{AAR040: Channel references a server that is not defined in the 'servers' object.}}
    messages:
      Notification:
        payload:
          type: object
operations:
  SendNotification:
    action: send
    channel:
      $ref: '#/channels/notifications'
    messages:
      - $ref: '#/channels/notifications/messages/Notification'

Compliant Solution (AsyncAPI 3)

asyncapi: '3.0.0'
info:
  title: Valid Server Reference
  version: '1.0.0'
servers:
  production:
    host: api.example.com
    protocol: wss
channels:
  notifications:
    address: notifications
    servers:
      - production
    messages:
      Notification:
        payload:
          type: object
operations:
  SendNotification:
    action: send
    channel:
      $ref: '#/channels/notifications'
    messages:
      - $ref: '#/channels/notifications/messages/Notification'
