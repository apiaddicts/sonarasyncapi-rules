Noncompliant Code Example (AsyncAPI 2)

asyncapi: '2.4.0'
info:
  title: My Async API
  version: '1.0.0'
channels:
  pet/created:
    subscribe:   # Noncompliant {{AAR051: The operationId must be present and follow camelCase naming convention}}
      summary: Receive pet created events.
      message:
        payload:
          type: object

Compliant Solution (AsyncAPI 2)

asyncapi: '2.4.0'
info:
  title: My Async API
  version: '1.0.0'
channels:
  pet/created:
    subscribe:
      operationId: solicitarBeca
      summary: Receive pet created events.
      message:
        payload:
          type: object

channels:

operations:

Compliant Solution (AsyncAPI 3)

asyncapi: '3.0.0'
info:
  title: My Async API
  version: '1.0.0'
channels:
  petCreated:
    address: pet/created
    messages:
      PetCreated:
        payload:
          type: object
operations:
  onPetCreated:
    action: receive
    operationId: solicitarBeca
    channel:
      $ref: '#/channels/petCreated'
    messages:
      - $ref: '#/channels/petCreated/messages/PetCreated'
