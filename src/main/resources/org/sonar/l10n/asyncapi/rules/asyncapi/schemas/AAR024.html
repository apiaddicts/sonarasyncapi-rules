Compliant Solution (AsyncAPI 2)

asyncapi: '2.4.0'
info:
  title: Message Validation
  version: '1.0.0'

  production:
    url: 'api.example.com'

channels:
  orders:
    subscribe:
      operationId: receiveOrder

          type: object

              type: string
