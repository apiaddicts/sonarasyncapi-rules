Compliant Solution (AsyncAPI 2)

    asyncapi: '2.4.0'
    info:
      title: Unique Operation ID Example API

    channels:
      /pets:
        description: Canal para operaciones relacionadas con mascotas.

          operationId: getPets

    protocol: wss

  events:
    address: events
    messages:
      Event:

      $ref: '#/channels/events'

Compliant Solution (AsyncAPI 3)

asyncapi: '3.0.0'
info:
  title: Operation Summaries
  version: 1.0.0
servers:
  production:
    host: api.example.com
    protocol: wss
channels:
  notifications:
    address: notifications
    messages:
      Notification:
        payload:
          type: object
operations:
  SendNotification:
    action: send
    channel:
      $ref: '#/channels/notifications'
    summary: Send a notification to users
    operationId: sendNotification
